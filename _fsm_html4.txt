<script>
(function(){
  var btn=document.getElementById('menuBtn'),ov=document.getElementById('fsmOverlay'),cl=document.getElementById('fsmClose');
  function openFSM(){if(ov){ov.classList.add('open');document.body.style.overflow='hidden';}}
  function closeFSM(){if(ov){ov.classList.remove('open');document.body.style.overflow='';}}
  if(btn)btn.addEventListener('click',openFSM);
  if(cl)cl.addEventListener('click',closeFSM);
  if(ov)ov.addEventListener('click',function(e){if(e.target===ov)closeFSM();});
  document.addEventListener('keydown',function(e){if(e.key==='Escape')closeFSM();});
})();
function toggleSide(el){var g=el.parentElement;document.querySelectorAll('.fsm-side-group.open').forEach(function(o){if(o!==g)o.classList.remove('open');});g.classList.toggle('open');}
</script>
